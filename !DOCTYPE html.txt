<script>

/*
==========================================
ITT ADD MEG A HELYES 11 JEGYŰ KÓDOT
==========================================
*/
const correctCode = "01234567890";

const inputs = document.querySelectorAll(".code-input");

/* Automatikus léptetés */
inputs.forEach((input, index) => {

    input.addEventListener("input", () => {

        input.value = input.value.replace(/[^0-9]/g, "");

        if (
            input.value.length === 1 &&
            index < inputs.length - 1
        ) {
            inputs[index + 1].focus();
        }

    });

    input.addEventListener("keydown", (e) => {

        if (
            e.key === "Backspace" &&
            input.value === "" &&
            index > 0
        ) {
            inputs[index - 1].focus();
        }

    });

});

/* Enter gombbal is ellenőrizhető */
document.addEventListener("keydown", function(e) {
    if (e.key === "Enter") {
        checkCode();
    }
});

function checkCode() {

    let enteredCode = "";

    inputs.forEach(input => {
        enteredCode += input.value;
    });

    const message = document.getElementById("message");

    if (enteredCode === correctCode) {

        message.style.color = "green";

        /*
        ==========================================
        ITT MÓDOSÍTHATOD A SIKER SZÖVEGET
        ==========================================
        */
        message.innerHTML =
        "Az alábbi jelszót vidd a kapuőrnek aki ha tehetné, most kápolna: Az Eucharisztia autópálya a mennyországba";

    } else {

        message.style.color = "red";

        /*
        ==========================================
        ITT MÓDOSÍTHATOD A HIBAÜZENETET
        ==========================================
        */
        message.innerHTML =
        "Hibás kód, próbáld újra!";

    }
}

</script>
